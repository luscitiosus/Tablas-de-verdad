var TZ = 'Atlantic/Canary', GRACIA_MS = 20000, N_EJ = 20;
var H_SES = ['sid','alumno','clase','correo','inicio','limite_ms','fin','estado','salidas','seg_fuera','comprobaciones','correcciones_vistas','resueltos','con_ayuda','resueltos_sin_ayuda','nota_sobre_10','dispositivo','device_id','ultimo_latido_ms'];
var H_EV = ['hora','sid','alumno','clase','tipo','detalle'];
var H_AL = ['hora','alumno','clase','tipo','detalle'];
var C = {sid:0,alumno:1,clase:2,correo:3,inicio:4,limite:5,fin:6,estado:7,salidas:8,fuera:9,checks:10,helps:11,solved:12,helped:13,sin:14,nota:15,disp:16,dev:17,beat:18};

function preparar() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var s = hoja_(ss, 'Sesiones', H_SES);
  s.getRange('M:N').setNumberFormat('@');
  hoja_(ss, 'Eventos', H_EV);
  hoja_(ss, 'Alertas', H_AL);
  if (!ss.getSheetByName('Config')) {
    var cf = ss.insertSheet('Config');
    cf.getRange(1, 1, 6, 3).setValues([
      ['CODIGO', 'tablas1', 'Código de sesión que das en clase (cámbialo en cada grupo)'],
      ['MINUTOS', 30, 'Duración global de la actividad, contada desde que el alumno pulsa Empezar'],
      ['MAX_SALIDAS', 3, 'Salidas de la pestaña a partir de las cuales se crea una alerta'],
      ['APERTURA', '', 'Opcional: fecha y hora de apertura (p. ej. 11/11/2026 09:00)'],
      ['CIERRE', '', 'Opcional: fecha y hora hasta la que se puede empezar'],
      ['ACTIVIDAD', 'tablas-verdad-1', 'Identificador informativo']]);
    cf.setColumnWidth(3, 480);
  }
}
function hoja_(ss, n, h) {
  var s = ss.getSheetByName(n);
  if (!s) { s = ss.insertSheet(n); s.getRange(1, 1, 1, h.length).setValues([h]).setFontWeight('bold'); s.setFrozenRows(1); }
  return s;
}
function cfg_() {
  var v = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Config').getRange(1, 1, 6, 2).getValues(), o = {};
  v.forEach(function (r) { o[r[0]] = r[1]; });
  return o;
}
function out_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
function doGet() { return out_({ ok: true, msg: 'Registro activo' }); }

function doPost(e) {
  var d;
  try { d = JSON.parse(e.postData.contents); } catch (x) { return out_({ ok: false, msg: 'Petición no válida' }); }
  var lock = LockService.getScriptLock();
  try { lock.waitLock(20000); } catch (x) { return out_({ ok: false, msg: 'Servidor ocupado, reintenta' }); }
  try {
    if (d.a === 'start') return out_(start_(d));
    if (d.a === 'resume') return out_(resume_(d));
    if (d.a === 'beat') return out_(beat_(d));
    return out_({ ok: false, msg: 'Acción desconocida' });
  } catch (err) {
    return out_({ ok: false, msg: 'Error: ' + err });
  } finally { lock.releaseLock(); }
}

function san_(s, n) {
  s = String(s == null ? '' : s).replace(/[\u0000-\u001f]/g, ' ').slice(0, n || 200);
  return /^[=+\-@]/.test(s) ? "'" + s : s;
}
function norm_(s) { return String(s).toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/\s+/g, ' ').trim(); }
function fmt_(ms) { return Utilities.formatDate(new Date(ms), TZ, 'yyyy-MM-dd HH:mm:ss'); }
function ses_() { return SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Sesiones'); }
function findRow_(sh, fn) {
  var n = sh.getLastRow();
  if (n < 2) return 0;
  var v = sh.getRange(2, 1, n - 1, H_SES.length).getValues();
  for (var i = 0; i < v.length; i++) if (fn(v[i])) return i + 2;
  return 0;
}
function alerta_(al, cl, tipo, det) {
  SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Alertas').appendRow([fmt_(Date.now()), al, cl, tipo, det]);
}
function lista_(a) {
  return (Array.isArray(a) ? a : []).map(Number)
    .filter(function (n) { return n >= 1 && n <= 99 && n % 1 === 0; })
    .filter(function (n, i, x) { return x.indexOf(n) === i; })
    .sort(function (a, b) { return a - b; });
}

function start_(d) {
  var c = cfg_(), now = Date.now();
  if (norm_(d.codigo || '') !== norm_(c.CODIGO)) return { ok: false, msg: 'Código de sesión incorrecto.' };
  if (c.APERTURA instanceof Date && now < c.APERTURA.getTime()) return { ok: false, msg: 'La actividad todavía no está abierta.' };
  if (c.CIERRE instanceof Date && now > c.CIERRE.getTime()) return { ok: false, msg: 'La ventana de la actividad ya ha cerrado.' };
  var nombre = san_(String(d.nombre || '').trim(), 80), clase = san_(d.clase, 30);
  if (nombre.length < 5 || !clase) return { ok: false, msg: 'Escribe tu nombre completo y elige tu clase.' };
  var key = norm_(nombre) + '|' + norm_(clase), sh = ses_();
  var r = findRow_(sh, function (v) { return norm_(v[C.alumno]) + '|' + norm_(v[C.clase]) === key; });
  if (r) return reanudar_(sh, r, d, now);
  var min = Number(c.MINUTOS) || 30, sid = Utilities.getUuid();
  sh.appendRow([sid, nombre, clase, san_(d.correo, 80), fmt_(now), now + min * 60000, '', 'ACTIVA', 0, 0, 0, 0, '', '', 0, 0, san_(d.ua, 120), san_(d.dev, 40), now]);
  return { ok: true, sid: sid, restante: min * 60, nombre: nombre, clase: clase };
}
function resume_(d) {
  var sh = ses_(), r = findRow_(sh, function (v) { return v[C.sid] === d.sid; });
  if (!r) return { ok: false, expired: true, msg: 'No se encuentra tu sesión. Entra de nuevo con tu nombre, clase y código.' };
  return reanudar_(sh, r, d, Date.now());
}
function reanudar_(sh, r, d, now) {
  var v = sh.getRange(r, 1, 1, H_SES.length).getValues()[0];
  var rest = Math.max(0, Math.floor((Number(v[C.limite]) - now) / 1000));
  if (v[C.estado] === 'ACTIVA' && rest <= 0) {
    sh.getRange(r, C.fin + 1, 1, 2).setValues([[fmt_(Number(v[C.limite])), 'FINALIZADA']]);
    v[C.estado] = 'FINALIZADA';
  }
  if (v[C.estado] !== 'ACTIVA') return { ok: false, expired: true, msg: 'La actividad ya está finalizada para ti (se agotó el tiempo o ya la enviaste).' };
  if (String(v[C.dev]) !== String(d.dev || '') && now - Number(v[C.beat]) < 90000)
    alerta_(v[C.alumno], v[C.clase], 'sesión simultánea', 'Reanudada desde otro dispositivo mientras seguía activa');
  sh.getRange(r, C.dev + 1, 1, 2).setValues([[san_(d.dev, 40), now]]);
  return { ok: true, sid: v[C.sid], restante: rest, nombre: v[C.alumno], clase: v[C.clase] };
}

function beat_(d) {
  var sh = ses_(), now = Date.now();
  var r = findRow_(sh, function (v) { return v[C.sid] === d.sid; });
  if (!r) return { ok: false, msg: 'Sesión no encontrada' };
  var v = sh.getRange(r, 1, 1, H_SES.length).getValues()[0];
  var lim = Number(v[C.limite]), rest = Math.max(0, Math.floor((lim - now) / 1000));
  if (v[C.estado] !== 'ACTIVA' || now > lim + GRACIA_MS) {
    if (v[C.estado] === 'ACTIVA') sh.getRange(r, C.fin + 1, 1, 2).setValues([[fmt_(lim), 'FINALIZADA']]);
    return { ok: true, restante: 0, fin: true };
  }
  var s = d.state || {}, ev = (d.events || []).slice(0, 60), al = v[C.alumno], cl = v[C.clase];
  if (ev.length) {
    var rows = ev.map(function (x) { return [fmt_(Number(x.t) || now), d.sid, al, cl, san_(x.tipo, 30), san_(x.det, 250)]; });
    var es = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Eventos');
    es.getRange(es.getLastRow() + 1, 1, rows.length, 6).setValues(rows);
  }
  var sal = Math.max(0, Math.floor(Number(s.salidas) || 0)), maxS = Number(cfg_().MAX_SALIDAS) || 3;
  var solved = lista_(s.solved), helped = lista_(s.helped);
  var sin = solved.filter(function (n) { return helped.indexOf(n) < 0; }).length;
  if (String(v[C.dev]) !== String(d.dev || '') && now - Number(v[C.beat]) < 90000)
    alerta_(al, cl, 'sesión simultánea', 'Latidos desde dos dispositivos distintos');
  if (sal >= maxS && Number(v[C.salidas]) < maxS) alerta_(al, cl, 'salidas de la pestaña', sal + ' salidas');
  var fin = !!d.fin || now >= lim;
  sh.getRange(r, C.salidas + 1, 1, 8).setValues([[sal, Math.round(Number(s.fuera) || 0), Math.floor(Number(s.checks) || 0), helped.length, solved.join(', '), helped.join(', '), sin, Math.round(sin / N_EJ * 100) / 10]]);
  sh.getRange(r, C.dev + 1, 1, 2).setValues([[san_(d.dev, 40), now]]);
  if (fin) sh.getRange(r, C.fin + 1, 1, 2).setValues([[fmt_(now), 'FINALIZADA']]);
  return { ok: true, restante: rest, fin: fin };
}
